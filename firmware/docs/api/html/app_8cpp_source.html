230static esp_err_t imu_init(void)
231{
232 if (imu_ready) {
233 return ESP_OK;
234 }
235 esp_err_t err = ESP_OK;
236 if (imu_dev == NULL) {
237 i2c_master_bus_config_t bus_config = {};
238 bus_config.i2c_port = I2C_NUM_0;
239 bus_config.sda_io_num = APP_I2C_SDA_GPIO;
240 bus_config.scl_io_num = APP_I2C_SCL_GPIO;
241 bus_config.clk_source = I2C_CLK_SRC_DEFAULT;
242 bus_config.glitch_ignore_cnt = 7;
243 bus_config.flags.enable_internal_pullup = true;
244 err = i2c_new_master_bus(&bus_config, &imu_bus);
245 if (err != ESP_OK) {
246 ESP_LOGE(TAG, "I2C bus initialization failed: %s", esp_err_to_name(err));
247 return err;
248 }
249 i2c_device_config_t device_config = {};
250 device_config.dev_addr_length = I2C_ADDR_BIT_LEN_7;
251 device_config.device_address = APP_MPU6050_ADDRESS;
252 device_config.scl_speed_hz = APP_I2C_FREQ_HZ;
253 err = i2c_master_bus_add_device(imu_bus, &device_config, &imu_dev);
254 if (err != ESP_OK) {
255 ESP_LOGE(TAG, "MPU6050 device attach failed: %s", esp_err_to_name(err));
256 return err;
257 }
258 }
259 uint8_t who = 0;
260 err = imu_read(0x75, &who, 1);
261 if (err != ESP_OK || (who != 0x68 && who != 0x69)) {
262 ESP_LOGE(TAG, "MPU6050 WHO_AM_I failed: err=%s value=0x%02x",
263 esp_err_to_name(err), who);
264 return err == ESP_OK ? ESP_ERR_NOT_FOUND : err;
265 }
266 const uint8_t init_regs[][2] = {{0x6B, 0x00}, {0x1A, 0x03}, {0x1B, 0x00}, {0x1C, 0x00}};
267 for (size_t i = 0; i < sizeof(init_regs) / sizeof(init_regs[0]); ++i) {
268 err = imu_write(init_regs[i][0], init_regs[i][1]);
269 if (err != ESP_OK) {
270 ESP_LOGE(TAG, "MPU6050 register 0x%02x write failed: %s",
271 init_regs[i][0], esp_err_to_name(err));
272 return err;
273 }
274 }
275 imu_ready = true;
276 ESP_LOGI(TAG, "MPU6050 ready at 0x%02x, SDA=%d SCL=%d", APP_MPU6050_ADDRESS,
277 APP_I2C_SDA_GPIO, APP_I2C_SCL_GPIO);
278 return ESP_OK;
279}

294static int classify_turbulence(const imu_sample_t *s)
295{
296 if (classifier_window_count + EI_CLASSIFIER_RAW_SAMPLES_PER_FRAME > EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE) {
297 classifier_window_count = 0;
298 }
299 classifier_window[classifier_window_count++] = s->roll;
300 classifier_window[classifier_window_count++] = s->pitch;
301 classifier_window[classifier_window_count++] = s->yaw;
302 if (classifier_window_count < EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE) {
303 return last_classifier_state;
304 }
305
306 signal_t signal = {};
307 signal.total_length = classifier_window_count;
308 signal.get_data = ei_signal_get_data;
309 ei_impulse_result_t result = {};
310 EI_IMPULSE_ERROR err = run_classifier(&signal, &result, false);
311 classifier_window_count = 0;
312 if (err != EI_IMPULSE_OK) {
313 ESP_LOGW(TAG, "Edge Impulse inference failed: %d", (int)err);
314 return last_classifier_state;
315 }
316
317 float best = -1.0f;
318 int state = last_classifier_state;
319 for (size_t i = 0; i < EI_CLASSIFIER_LABEL_COUNT; ++i) {
320 const char *label = result.classification[i].label;
321 const float value = result.classification[i].value;
322 if (value > best) {
323 best = value;
324 if (strstr(label, "severe") != NULL) {
325 state = 2;
326 } else if (strstr(label, "light") != NULL) {
327 state = 1;
328 } else {
329 state = 0;
330 }
331 }
332 }
333 last_classifier_state = state;
334 ESP_LOGI(TAG, "Edge Impulse state=%d confidence=%.3f", state, best);
335 return state;
336}

338static void sensor_task(void *arg)
339{
340 (void)arg;
341 TickType_t wake = xTaskGetTickCount();
342 int errors = 0;
343 vTaskDelay(pdMS_TO_TICKS(100));
344 while (true) {
345 if (!imu_ready) {
346 static int64_t last_retry_us;
347 int64_t now = esp_timer_get_time();
348 if (now - last_retry_us >= 1000000) {
349 last_retry_us = now;
350 ESP_LOGW(TAG, "IMU unavailable; retrying initialization");
351 if (imu_init() != ESP_OK) {
352 vTaskDelayUntil(&wake, pdMS_TO_TICKS(APP_SENSOR_PERIOD_MS));
353 continue;
354 }
355 }
356 }
357 uint8_t raw[14];
358 esp_err_t err = imu_read(0x3B, raw, sizeof(raw));
359 if (err != ESP_OK) {
360 errors++;
361 if ((errors % 50) == 1) {
362 ESP_LOGW(TAG, "I2C IMU read failed (%d): %s", errors, esp_err_to_name(err));
363 }
364 vTaskDelayUntil(&wake, pdMS_TO_TICKS(APP_SENSOR_PERIOD_MS));
365 continue;
366 }
367 imu_sample_t s = {};
368 s.ax = 0; s.ay = 0; s.az = 0; s.gx = 0; s.gy = 0; s.gz = 0;
369 s.roll = 0.0f; s.pitch = 0.0f; s.yaw = 0.0f; s.state = 0; s.timestamp_us = 0;
370 s.ax = (int16_t)((raw[0] << 8) | raw[1]); s.ay = (int16_t)((raw[2] << 8) | raw[3]);
371 s.az = (int16_t)((raw[4] << 8) | raw[5]); s.gx = (int16_t)((raw[8] << 8) | raw[9]);
372 s.gy = (int16_t)((raw[10] << 8) | raw[11]); s.gz = (int16_t)((raw[12] << 8) | raw[13]);
373 float ax = s.ax / 16384.0f, ay = s.ay / 16384.0f, az = s.az / 16384.0f;
374 float accel_roll = atan2f(ay, az) * 57.29578f;
375 float accel_pitch = atan2f(-ax, sqrtf(ay * ay + az * az)) * 57.29578f;
376 float gx = s.gx / 131.0f, gy = s.gy / 131.0f, gz = s.gz / 131.0f;
377 if (!filter_ready) { latest.roll = accel_roll; latest.pitch = accel_pitch; latest.yaw = 0; filter_ready = true; }
378 else {
379 latest.roll = APP_COMPLEMENTARY_ALPHA * (latest.roll + gx * 0.02f) + (1.0f - APP_COMPLEMENTARY_ALPHA) * accel_roll;
380 latest.pitch = APP_COMPLEMENTARY_ALPHA * (latest.pitch + gy * 0.02f) + (1.0f - APP_COMPLEMENTARY_ALPHA) * accel_pitch;
381 latest.yaw += gz * 0.02f;
382 if (latest.yaw > 180) {
383 latest.yaw -= 360;
384 }
385 if (latest.yaw < -180) {
386 latest.yaw += 360;
387 }
388 }
389 s.roll = latest.roll; s.pitch = latest.pitch; s.yaw = latest.yaw; s.timestamp_us = esp_timer_get_time();
390 s.state = classify_turbulence(&s);
391 if (xSemaphoreTake(sample_mutex, pdMS_TO_TICKS(5)) == pdTRUE) {
392 latest = s;
393 xSemaphoreGive(sample_mutex);
394 }
395 status_led_set_state(s.state);
396 if (s.state == 2 && xQueueSend(alert_queue, &s, 0) != pdTRUE) {
397 ESP_LOGW(TAG, "Alert queue full");
398 }
399 printf("%lld,%.2f,%.2f,%.2f\n", s.timestamp_us / 1000, s.roll, s.pitch, s.yaw);
400 vTaskDelayUntil(&wake, pdMS_TO_TICKS(APP_SENSOR_PERIOD_MS));
401 }
402}
