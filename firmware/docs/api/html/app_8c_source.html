161 s.az = (int16_t)((raw[4] << 8) | raw[5]); s.gx = (int16_t)((raw[8] << 8) | raw[9]);
162 s.gy = (int16_t)((raw[10] << 8) | raw[11]); s.gz = (int16_t)((raw[12] << 8) | raw[13]);
163 float ax = s.ax / 16384.0f, ay = s.ay / 16384.0f, az = s.az / 16384.0f;
164 float accel_roll = atan2f(ay, az) * 57.29578f;
165 float accel_pitch = atan2f(-ax, sqrtf(ay * ay + az * az)) * 57.29578f;
166 float gx = s.gx / 131.0f, gy = s.gy / 131.0f, gz = s.gz / 131.0f;
167 if (!filter_ready) { latest.roll = accel_roll; latest.pitch = accel_pitch; latest.yaw = 0; filter_ready = true; }
168 else {
169 latest.roll = APP_COMPLEMENTARY_ALPHA * (latest.roll + gx * 0.02f) + (1.0f - APP_COMPLEMENTARY_ALPHA) * accel_roll;
170 latest.pitch = APP_COMPLEMENTARY_ALPHA * (latest.pitch + gy * 0.02f) + (1.0f - APP_COMPLEMENTARY_ALPHA) * accel_pitch;
171 latest.yaw += gz * 0.02f;
172 if (latest.yaw > 180) {
173 latest.yaw -= 360;
174 }
175 if (latest.yaw < -180) {
176 latest.yaw += 360;
177 }
178 }
179 s.roll = latest.roll; s.pitch = latest.pitch; s.yaw = latest.yaw; s.timestamp_us = esp_timer_get_time();
180 s.state = classify_turbulence(&s);

183 }
